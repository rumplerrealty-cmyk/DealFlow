Supabase Setup Instructions

1. Create tables

Go to Supabase → SQL Editor → New query and run this:

create table if not exists clients (
  id uuid primary key default gen_random_uuid(),
  name text not null,
  email text,
  phone text,
  notes text,
  created_at timestamptz default now()
);

create table if not exists deals (
  id uuid primary key default gen_random_uuid(),
  property text not null,
  client_id uuid references clients(id) on delete set null,
  status text default 'Offer Sent',
  closing date,
  notes text,
  created_at timestamptz default now()
);

create table if not exists tasks (
  id uuid primary key default gen_random_uuid(),
  title text not null,
  due date,
  deal_id uuid references deals(id) on delete set null,
  done boolean default false,
  notes text,
  created_at timestamptz default now()
);

create table if not exists files (
  id uuid primary key default gen_random_uuid(),
  deal_id uuid references deals(id) on delete cascade,
  name text not null,
  path text not null,
  size bigint,
  created_at timestamptz default now()
);

create table if not exists settings (
  key text primary key,
  value text
);

insert into settings (key, value) values ('team_password', 'dealflow2026')
on conflict (key) do nothing;

2. Create Storage bucket

Go to Storage → New bucket → Name it deal-files → Make it Public (or private + policies).

3. Enable RLS (simple open rules for now)

alter table clients enable row level security;
alter table deals enable row level security;
alter table tasks enable row level security;
alter table files enable row level security;
alter table settings enable row level security;

create policy "Allow all" on clients for all using (true) with check (true);
create policy "Allow all" on deals for all using (true) with check (true);
create policy "Allow all" on tasks for all using (true) with check (true);
create policy "Allow all" on files for all using (true) with check (true);
create policy "Allow all" on settings for all using (true) with check (true);

These open rules are fine for a private team tool. You can tighten them later.
